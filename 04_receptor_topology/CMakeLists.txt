# 04_receptor_topology/CMakeLists.txt
cmake_minimum_required(VERSION 3.20)

set(RECEPTOR_CONF_GRO "${CMAKE_CURRENT_BINARY_DIR}/conf.gro")
set(RECEPTOR_TOPOL_TOP "${CMAKE_CURRENT_BINARY_DIR}/topol.top")
set(RECEPTOR_POSRE_ITP "${CMAKE_CURRENT_BINARY_DIR}/posre.itp")

add_custom_command(
    OUTPUT "${RECEPTOR_CONF_GRO}" "${RECEPTOR_TOPOL_TOP}" "${RECEPTOR_POSRE_ITP}"
    COMMAND ${GMX_EXECUTABLE} pdb2gmx -f "${RECEPTOR_FIXED_PDB}" -o "${RECEPTOR_CONF_GRO}" -p "${RECEPTOR_TOPOL_TOP}" -i "${RECEPTOR_POSRE_ITP}" -water tip3p -ff amber99sb-ildn -ignh
    DEPENDS 03_rebuild_atoms "${RECEPTOR_FIXED_PDB}" "${GMX_EXECUTABLE}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    COMMENT "Processing receptor topology with GROMACS pdb2gmx"
)

add_custom_target(04_receptor_topology DEPENDS "${RECEPTOR_CONF_GRO}" "${RECEPTOR_TOPOL_TOP}" "${RECEPTOR_POSRE_ITP}")
set(RECEPTOR_CONF_GRO "${RECEPTOR_CONF_GRO}" PARENT_SCOPE)
set(RECEPTOR_TOPOL_TOP "${RECEPTOR_TOPOL_TOP}" PARENT_SCOPE)
set(RECEPTOR_POSRE_ITP "${RECEPTOR_POSRE_ITP}" PARENT_SCOPE)
